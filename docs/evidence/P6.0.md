## 4. Experiment (c): both versions alive, custom Unpickler.find_class

```python
"""(c) keep BOTH edtlib versions alive at once; custom Unpickler maps 'devicetree.*' to a chosen module set. No purge needed between loads."""
import sys, pickle, importlib, time
V = {"old": (r"C:\Masters\26020_RTOS4\zephyrproject\zephyr\scripts\dts\python-devicetree\src",
             r"C:\Masters\26020_RTOS4\zephyrproject\apps\lab1\build_same54_xpro\zephyr\edt.pickle"),
     "new": (r"C:\Masters\zephyr-main\zephyr\scripts\dts\python-devicetree\src",
             r"C:\Users\M91221\AppData\Local\Temp\dtx_main_sg41\zephyr\edt.pickle")}
MODS = {}   # version -> {"devicetree.edtlib": module, ...}
def import_version(tag):
    src = V[tag][0]
    for k in [k for k in sys.modules if k == "devicetree" or k.startswith("devicetree.")]: del sys.modules[k]
    sys.path.insert(0, src)
    try:
        for n in ("devicetree", "devicetree._private", "devicetree.dtlib", "devicetree.grutils", "devicetree.edtlib"):
            importlib.import_module(n)
        MODS[tag] = {k: v for k, v in sys.modules.items() if k == "devicetree" or k.startswith("devicetree.")}
    finally:
        sys.path.remove(src)
        for k in list(MODS[tag]): sys.modules.pop(k, None)   # nothing named devicetree stays global
class U(pickle.Unpickler):
    def __init__(self, f, tag): super().__init__(f); self.tag = tag
    def find_class(self, module, name):
        if module == "devicetree" or module.startswith("devicetree."):
            return getattr(MODS[self.tag][module], name)
        return super().find_class(module, name)
def load(tag):
    with open(V[tag][1], "rb") as f: return U(f, tag).load()
t = time.time(); import_version("old"); import_version("new"); print("import both: %.2fs" % (time.time()-t))
print("global sys.modules has devicetree:", any(k.startswith("devicetree") for k in sys.modules))
a = load("old"); b = load("new")
print("distinct class objects:", type(a).__module__, type(a) is not type(b), MODS["old"]["devicetree.edtlib"].__file__ != MODS["new"]["devicetree.edtlib"].__file__)

          len(e.compat2nodes), isinstance(n, MODS[tag]["devicetree.edtlib"].Node), isinstance(n, MODS["old" if tag=="new" else "new"]["devicetree.edtlib"].Node))
    pc = [x for x in e.nodes if x.pinctrls]
    print(tag, "nodes with pinctrls:", len(pc), (pc[0].path, [(p.name, [s.path for s in p.conf_nodes][:1]) for p in pc[0].pinctrls][:1]) if pc else None)
    print(tag, "label2node", len(e.label2node), "chosen", sorted(e.chosen_nodes)[:3], "alias", len(e._dt.alias2node))
```
