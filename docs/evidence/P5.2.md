Generator source (`gen_tables.py`):

```python
"""Generate the data tables of atmel_sam0_to_mchp_g1.yaml from the workspace headers
and verify the MCLK bus mapping against both SoC dtsi files.

Usage: py gen_tables.py <workspace> [--verify-only]
Prints the YAML text of the three generated tables on stdout, verification on stderr.
"""
import re, sys, collections
ws = sys.argv[1]
Z = ws + r"\zephyr"
hdr = open(Z + r"\include\zephyr\dt-bindings\clock\mchp_sam_d5x_e5x_clock.h", encoding="utf8").read()

def derive(t, bus, bit, gp, inst):
    return (t << 26) | (bus << 20) | (bit << 14) | (gp << 8) | inst

mclk, gclk = {}, collections.defaultdict(list)
for m in re.finditer(r"#define\s+(CLOCK_MCHP_(MCLKPERIPH|GCLKPERIPH)_ID_\w+)\s*\\?\s*MCHP_CLOCK_DERIVE_ID\(([^)]*)\)", hdr):
    name, kind, args = m.group(1), m.group(2), [int(x, 0) for x in m.group(3).split(",")]
    val = derive(*args)
    if kind == "MCLKPERIPH":
        mclk[f"{args[1]}:{args[2]}"] = [name, val]
    else:
        gclk[str(args[3])].append([name, val])

# pin macro renames
def pins(p):
    t = open(p, encoding="utf8").read()
    return {m.group(1): re.sub(r"\s+", "", m.group(2)) for m in
            re.finditer(r"#define\s+(P[A-D]\d+[A-P]_\w+)\s*\\?\s*(?:SAM|MCHP)_PINMUX\(([^)]*)\)", t)}
old = pins(ws + r"\modules\hal\atmel\include\dt-bindings\pinctrl\same54p-pinctrl.h")
new = pins(ws + r"\modules\hal\microchip\include\dt-bindings\sam\sam_d5x_e5x\atsame54\atsame54p20a-pinctrl.h")
oo = sorted(set(old) - set(new)); nn = sorted(set(new) - set(old))
bypre = collections.defaultdict(list)
for n in nn:
    bypre[n.split("_")[0]].append(n)
ren = {}
for o in oo:
    pre, rest = o.split("_", 1)
    cand = bypre.get(pre, [])
    tgt = None
    if len(cand) == 1:
        tgt = cand[0]
    else:
        m = re.fullmatch(r"PTC_XY(\d+)", rest)
        if m and f"{pre}_ADC0_X{m.group(1)}" in cand:
            tgt = f"{pre}_ADC0_X{m.group(1)}"
        m = re.fullmatch(r"ANAREF_(VREF\w)", rest)
        if m and f"{pre}_ADC0_{m.group(1)}" in cand:
            tgt = f"{pre}_ADC0_{m.group(1)}"
    ren[o] = tgt
    if tgt and old[o] != new[tgt]:
        print("ENCODING DIFFERS", o, tgt, file=sys.stderr)

# verify the MCLK offset -> bus mapping against both SoC dtsi
BUS = {0x10: 0, 0x14: 1, 0x18: 2, 0x1C: 3, 0x20: 4}
def nodes(path, rx):
    t = open(path, encoding="utf8").read()
    d = {}
    for m in re.finditer(r"(\w+): \w+@([0-9a-f]+) \{(.*?)\n\t\t\};", t, re.S):
        c = re.search(rx, m.group(3), re.S)
        if c:
            d[m.group(2)] = c
    return d
o = nodes(Z + r"\dts\arm\atmel\samd5x.dtsi", r"<&mclk (0x[0-9a-f]+) (\d+)>")
C = Z + r"\dts\arm\microchip\sam\sam_d5x_e5x\common\samd5xe5x"
nw = {}
for sfx in ("", "_j", "_n", "_p"):
    nw.update(nodes(C + sfx + ".dtsi", r"<&mclkperiph (\w+)>"))
buses = set()
macro2key = {v[0].replace("CLOCK_MCHP_MCLKPERIPH_ID_", ""): k for k, v in mclk.items()}
ok = bad = 0
for addr, c in o.items():
    if addr in nw:
        key = f"{BUS[int(c.group(1), 16)]}:{int(c.group(2))}"
        want = macro2key.get(nw[addr].group(1).replace("CLOCK_MCHP_MCLKPERIPH_ID_", ""))
        if key == want:
            ok += 1
            buses.add(c.group(1))
        else:
            bad += 1
            print("MISMATCH", addr, key, want, file=sys.stderr)
print(f"mclk bus mapping check: {ok} matching nodes, {bad} mismatches, offsets covered: {sorted(buses)}", file=sys.stderr)
# same check for gclk ids
og = nodes(Z + r"\dts\arm\atmel\samd5x.dtsi", r"<&gclk (\d+)>")
ng = nodes(Z + r"\dts\arm\microchip\sam\sam_d5x_e5x\common\samd5xe5x.dtsi", r"<&gclkperiph (\w+)>")
ok = bad = 0
for addr, c in og.items():
    if addr in ng:
        names = [x[0] for x in gclk.get(str(int(c.group(1))), [])]
        if ng[addr].group(1) in names:
            ok += 1
        else:
            bad += 1
            print("GCLK MISMATCH", addr, c.group(1), ng[addr].group(1), names, file=sys.stderr)
print(f"gclk id mapping check: {ok} matching nodes, {bad} mismatches", file=sys.stderr)
print(f"pin renames: {len(ren)} old-only names, {sum(1 for v in ren.values() if v)} with target, "
      f"{sum(1 for v in ren.values() if not v)} without; mclk {len(mclk)}, gclk ids {len(gclk)}", file=sys.stderr)

if "--verify-only" not in sys.argv:
    print("  pin_renames:")
    for k in sorted(ren):
        print(f"    {k}: {ren[k] if ren[k] else 'null'}")
    print("  mclk_periph:   # \"bus:bit\" -> [macro, id]; bus 0=AHB 1=APBA 2=APBB 3=APBC 4=APBD")
    for k in sorted(mclk, key=lambda s: tuple(map(int, s.split(":")))):
        print(f"    \"{k}\": [{mclk[k][0]}, {mclk[k][1]}]")
    print("  gclk_periph:   # gclk channel -> [[macro, id], ...]")
    for k in sorted(gclk, key=int):
        print(f"    \"{k}\": [" + ", ".join(f"[{n}, {v}]" for n, v in gclk[k]) + "]")
```
