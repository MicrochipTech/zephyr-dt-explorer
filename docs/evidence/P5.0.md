## A. Enumerator source (enum2.py)

```python
# Enumerate ALL properties of a Zephyr binding YAML (read-only), resolving include: chains
# (with property-allowlist / property-blocklist) and child-binding levels.
# Each property is listed once per level with its origin: "own" or "inh:<file>" (file = included YAML).
# usage: py enum2.py <bindings_root> <binding-yaml-basename>...
import sys, os, yaml

ROOT = sys.argv[1]
INDEX = {}
for dp, _, fns in os.walk(ROOT):
    for fn in fns:
        if fn.endswith('.yaml'):
            INDEX.setdefault(fn, os.path.join(dp, fn))


def load(path):
    with open(path, encoding='utf-8') as f:
        return yaml.safe_load(f) or {}


def rel(p):
    return os.path.relpath(p, ROOT).replace('\\', '/')


def norm_includes(d):
    inc = d.get('include')
    if inc is None:
        return []
    if isinstance(inc, (str, dict)):
        inc = [inc]
    out = []
    for e in inc:
        out.append({'name': e} if isinstance(e, str) else dict(e))
    return out


def filtered(props, allow, block):
    if allow is not None:
        props = {k: v for k, v in props.items() if k in allow}
    if block is not None:
        props = {k: v for k, v in props.items() if k not in block}
    return props


def level_props(path, origin_top, allow=None, block=None, seen=None):
    """Properties of the top level of the binding at path, inherited ones included.
    Returns ordered dict name -> (spec, origin)."""
    seen = seen or set()
    d = load(path)
    chain = []  # (origin, props) base -> derived
    for e in norm_includes(d):
        ip = INDEX.get(os.path.basename(e['name']))
        if ip is None:
            chain.append(('MISSING:' + e['name'], {}))
            continue
        sub = level_props(ip, rel(ip), e.get('property-allowlist'), e.get('property-blocklist'), seen)
        chain.extend(sub)
    own = d.get('properties') or {}
    chain.append((origin_top, own))
    final = {}
    for origin, props in chain:
        for k, spec in props.items():
            final[k] = (spec or {}, origin)
    return filtered_result(final, allow, block)


def filtered_result(final, allow, block):
    if allow is not None:
        final = {k: v for k, v in final.items() if k in allow}
    if block is not None:
        final = {k: v for k, v in final.items() if k not in block}
    return final


def chain_list(path, origin, allow=None, block=None):
    """Like level_props but returns list of (origin, props) (for includes)."""
    d = load(path)
    out = []
    for e in norm_includes(d):
        ip = INDEX.get(os.path.basename(e['name']))
        if ip is None:
            continue
        out.extend(chain_list(ip, rel(ip), e.get('property-allowlist'), e.get('property-blocklist')))
    out.append((origin, d.get('properties') or {}, allow, block))
    return out


def merged_top(path):
    """Return ordered dict name -> (spec, origin) for top-level props with includes."""
    final = {}
    for origin, props, allow, block in chain_list(path, rel(path)):
        props = filtered(props, allow, block)
        for k, spec in props.items():
            final[k] = (spec or {}, origin)
    return final


def child_levels(d, path_label, out):
    """Collect child-binding levels declared in dict d (YAML of a binding or a child-binding)."""
    cb = d.get('child-binding')
    if cb is None:
        return
    label = path_label + '/child-binding'
    own = cb.get('properties') or {}
    inc_props = {}
    for e in norm_includes(cb):
        ip = INDEX.get(os.path.basename(e['name']))
        if ip is None:
            continue
        for origin, props, allow, block in chain_list(ip, rel(ip), e.get('property-allowlist'), e.get('property-blocklist')):
            for k, spec in filtered(props, allow, block).items():
                inc_props[k] = (spec or {}, origin)
    merged = dict(inc_props)
    for k, spec in own.items():
        merged[k] = (spec or {}, 'own')
    out.append((label, merged, bool(norm_includes(cb))))
    child_levels(cb, label, out)


if __name__ == '__main__':
    for arg in sys.argv[2:]:
        path = INDEX.get(os.path.basename(arg))
        if path is None:
            print('## NOT FOUND', arg)
            continue
        d = load(path)
        top = merged_top(path)
        own_top = set((d.get('properties') or {}).keys())
        print(f'## {rel(path)}  compatible={d.get("compatible")}  include={[e["name"] for e in norm_includes(d)]}')
        print(f'### level: top (properties)  count={len(top)}')
        for k, (spec, origin) in top.items():
            tag = 'own' if k in own_top else 'inh:' + origin
            print(f'  {k}\ttype={spec.get("type", "-")}\trequired={"yes" if spec.get("required") else "no"}\t{tag}')
        levels = []
        child_levels(d, 'child', levels)
        total = len(top)
        for label, merged, has_inc in levels:
            print(f'### level: {label}  count={len(merged)}')
            for k, (spec, origin) in merged.items():
                print(f'  {k}\ttype={spec.get("type", "-")}\trequired={"yes" if spec.get("required") else "no"}\t{origin}')
            total += len(merged)
        print(f'TOTAL-DECLARED-PROPERTIES {total}\n')
```
