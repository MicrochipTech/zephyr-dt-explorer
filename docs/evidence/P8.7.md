### Design

All board-level nodes are ONE draft node of class `board` with label `{/}` (no instance, no pins, no compatible).
`render_overlay` writes it as the node override `&{/} { ... };` (dtc's path reference). Its `DraftChild`ren are the
containers `user_leds` (`gpio-leds`), `user_keys` (`gpio-keys`), `user_pwm_leds` (`pwm-leds`) with one child per
LED / button / PWM LED (name = label), plus the children `aliases` and `chosen`.

* GPIO cell: `<&portd 10 flags>`; controller label and pin come from "pad" through the same mapping that
  `core.pins` uses for the pads existing nodes occupy (`gpio_pad_map` inverts `pins._gpio_pad`). The pad itself is
  not stored; so `validate`'s existing `pin_conflict` covers the new uses without any new code. Flags are integers
  (active low 1, pull-up 16, pull-down 32) because a draft file cannot hold the symbolic names.
* PWM LED: `<&tcc0 channel period_ns>` (a third `0` flags cell if the controller has `#pwm-cells = <3>`), for the PWM
  controllers that exist in the build (`#pwm-cells`). The PWM pin is the controller's own pinctrl.
* `aliases` / `chosen` hold string paths (`"/user_leds/led_user"`), valid in both blocks for dtc and edtlib; a
  phandle in `<>` would not be a valid alias. Targets: any labelled node of the build, or a new board node.
* The board's own `/leds`, `/buttons`, `/aliases` stay untouched (new containers next to them, dtc merges
  `aliases` / `chosen`).
* Port to board… does not carry the board node over (`no_node` for the label `{/}`); listed under Limits in GUI.md.

```python
"""P8.7 acceptance: board-level nodes on a COPY of _soltest/apps/lab1_leer."""
import os
import shutil
import sys
import tempfile
from pathlib import Path

sys.path.insert(0, os.getcwd())
from dt_explorer.app import load_plugins
from dt_explorer.core import board_nodes as B
from dt_explorer.core import verify
from dt_explorer.core.build import BuildModel
from dt_explorer.ui import configure as C
from dt_explorer.ui.draft_tab import kconfig_lines

load_plugins()
src = Path(os.environ["DTX_RTOS4_WS"]) / "_soltest" / "apps" / "lab1_leer"
work = Path(tempfile.gettempdir()) / "dtx_p87"
shutil.rmtree(work, ignore_errors=True)
app = work / "lab1_leer"
shutil.copytree(src, app)
build = app / "build_same54_xpro"
print("copy:", build)

m = BuildModel(build)
s = C.ConfigSession(m)          # session.save() is never called: the draft stays in memory
pad_map = B.gpio_pad_map(m, s.device)
used = s.used_by()
free = [p for p in sorted(pad_map) if not used.get(p)]
print("GPIO pads mapped:", len(pad_map), "| free:", len(free), "| used by the build:", sorted(p for p in used if p in pad_map))
led_pad, key_pad = "PD10", "PD11"
assert led_pad in free and key_pad in free, (free[:20])
print("LED pad", led_pad, "->", pad_map[led_pad], "| key pad", key_pad, "->", pad_map[key_pad])

pwm = B.pwm_controllers(m)
print("PWM controllers in the build:", pwm)

spec = B.BoardSpec(
    entries=[B.Entry(B.LEDS, "led_user", pad=led_pad, active_low=True, text="User LED"),
             B.Entry(B.KEYS, "btn_user", pad=key_pad, active_low=True, pull="pull-up", text="User button", code=11),
             B.Entry(B.PWM_LEDS, "pwm_user", ctrl="tcc0", channel=1, period_ns=20_000_000)],
    aliases={"led1": "/user_leds/led_user", "sw1": "/user_keys/btn_user", "pwm-led1": "/user_pwm_leds/pwm_user"},
    chosen={"dtx,user-led": "/user_leds/led_user", "zephyr,display-led": "/user_pwm_leds/pwm_user"})
problems = B.check_spec(spec, model=m, pad_map=pad_map, other_users={**used, **s.draft_pads()},
                        existing_aliases=set(m.edt._dt.alias2node), pwm_ctrls={p["label"]: p for p in pwm})
print("spec problems:", [str(p) for p in problems])
B.apply_spec(s.draft, spec, pad_map, {p["label"]: len(p["cells"]) for p in pwm})
assert B.read_spec(s.draft, pad_map) == spec
s.findings = None
print("draft nodes:", [(n.label, n.cls, B.summary(n)) for n in s.draft.nodes])
overlay = s.overlay()
print("\n--- overlay ---\n" + overlay)
findings = s.check()
print("--- validate findings (%d) ---" % len(findings))
for f in findings:
    print(f.severity, f.kind, f.node, f.message)
print("errors:", [f for f in findings if f.severity == "error"])
conf = "\n".join(kconfig_lines(s.kconfig())) + "\n"
print("kconfig lines:", repr(conf))
res = verify.verify(build, overlay, conf, on_line=None, scratch_root=work / "verify", timeout=600)
print("--- Verify with west ---")
print("rc:", res.returncode, "ok:", res.ok, "duration: %.1f s" % res.duration)
print("errors:", res.errors)
print("warnings:", res.warnings)
print("cmd:", " ".join(res.cmd))
print("log tail:\n" + "\n".join(res.log_tail[-8:]))

# round trip through the JSON draft form
from dt_explorer.core.draft import Draft
d2 = Draft.from_dict(s.draft.to_dict())
spec2 = B.read_spec(d2, pad_map)
print("\nround trip equal:", spec2 == spec, "| entries:", [(e.kind, e.name, e.pad) for e in spec2.entries])

# an LED on a used pad -> conflict error
print("\n=== LED on a used pad (PC18 is led0 of the board) ===")
s2 = C.ConfigSession(m)
bad = B.BoardSpec(entries=[B.Entry(B.LEDS, "led_bad", pad="PC18", text="bad")])
print("spec problems:", [str(p) for p in B.check_spec(bad, model=m, pad_map=pad_map, other_users=s2.used_by())])
B.apply_spec(s2.draft, bad, pad_map)
print(s2.overlay())
for f in s2.check():
    print(f.severity, f.kind, f.node, f.message)
```

## 5. Notes

* The Quasar `title`/tooltip trap: an element with an `id=` prop and `.tooltip(...)` logs `Anchor: target "#cNN" not
  found` in the browser console (NiceGUI's tooltip anchors to the generated id). The existing Draft tab edit button
  was given an id for the tests, so its tooltip became the `title` prop.
* `ui.select(new_value_mode=...)` accepts a typed name in a browser; in the simulated tests a programmatic value
  outside the options is dropped, so the UI test picks a name from the suggestions.
