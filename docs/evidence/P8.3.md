`enum3.py` imports `enum2.py` (P5.0, unchanged, source in `docs/evidence/P5.0.md` section A) and adds the child-bindings of included files. Source of `enum3.py`:

```python
# enum3.py: P8.3 extension of the P5.0 enumerator (enum2.py, imported unchanged).
# Difference: child-binding levels are also collected from INCLUDED bindings
# (e.g. adc-controller.yaml -> channel@N, dac-controller.yaml, can-controller.yaml), which enum2.py did not do.
# Top-level output is identical to enum2.py. usage: py enum3.py <bindings_root> <binding-yaml-basename>...
import sys, os
import enum2
from enum2 import INDEX, load, rel, norm_includes, chain_list, filtered, merged_top


def cb_stack(d):
    """child-binding dicts of a binding, from its includes (in include order) and its own, base -> derived."""
    stack = []
    for e in norm_includes(d):
        ip = INDEX.get(os.path.basename(e['name']))
        if ip is not None:
            stack.extend(cb_stack(load(ip)))
    if isinstance(d.get('child-binding'), dict):
        stack.append(d['child-binding'])
    return stack


def cb_props(cb):
    merged = {}
    for e in norm_includes(cb):
        ip = INDEX.get(os.path.basename(e['name']))
        if ip is None:
            continue
        for origin, props, allow, block in chain_list(ip, rel(ip), e.get('property-allowlist'), e.get('property-blocklist')):
            for k, spec in filtered(props, allow, block).items():
                merged[k] = (spec or {}, origin)
    for k, spec in (cb.get('properties') or {}).items():
        merged[k] = (spec or {}, 'own')
    return merged


def levels(d):
    out = []
    stack = cb_stack(d)
    label = 'child'
    while stack:
        label += '/child-binding'
        merged = {}
        for cb in stack:
            merged.update(cb_props(cb))
        out.append((label, merged))
        stack = [cb['child-binding'] for cb in stack if isinstance(cb.get('child-binding'), dict)]
    return out


if __name__ == '__main__':
    for arg in sys.argv[2:]:
        path = INDEX.get(os.path.basename(arg))
        if path is None:
            print('## NOT FOUND', arg)
            continue
        d = load(path)
        top = merged_top(path)
        own_top = set((d.get('properties') or {}).keys())
        print(f'## {rel(path)}  compatible={d.get("compatible")}  include={[e["name"] for e in norm_includes(d)]}')
        print(f'### level: top (properties)  count={len(top)}')
        for k, (spec, origin) in top.items():
            tag = 'own' if k in own_top else 'inh:' + origin
            print(f'  {k}\ttype={spec.get("type", "-")}\trequired={"yes" if spec.get("required") else "no"}\t{tag}')
        total = len(top)
        for label, merged in levels(d):
            print(f'### level: {label}  count={len(merged)}')
            for k, (spec, origin) in merged.items():
                print(f'  {k}\ttype={spec.get("type", "-")}\trequired={"yes" if spec.get("required") else "no"}\t{origin}')
            total += len(merged)
        print(f'TOTAL-DECLARED-PROPERTIES {total}\n')
```

## F. Node presence in the SoC dtsi files

`presence.py`:

```python
# P8.3: which peripheral nodes (by compatible) exist in which SoC dtsi files.
# usage: py presence.py <label> <root-for-relative-names> <dtsi>...
import sys, re, os
label, root, *files = sys.argv[1:]
WANT = re.compile(r'(tc|tcc|adc|dac|eic|gmac|mdio|rtc|wdt|watchdog|dmac|trng|random|can|usb|sam-ethernet-controller)', re.I)
print(f'### {label}')
print('file:line\tnode header\tcompatible')
for f in files:
    hdr = None
    for i, line in enumerate(open(f, encoding='utf-8', errors='replace'), 1):
        m = re.match(r'\s*((?:[A-Za-z0-9_]+:\s*)?[A-Za-z0-9_,.+-]+(?:@[0-9a-fA-F]+)?)\s*\{', line)
        if m and 'compatible' not in line:
            hdr = m.group(1)
        c = re.search(r'compatible\s*=\s*"([^"]+)"', line)
        if c:
            comp = c.group(1)
            if re.search(r'(atmel,sam0?-(tc|tcc|adc|dac|eic|gmac|mdio|rtc|watchdog|dmac|trng|can|usb)|atmel,samd5x-dac|atmel,sam-(trng|mdio)|microchip,(tc|tcc|adc|dac|eic|gmac|rtc|wdt|dmac|trng|sam-ethernet)[a-z0-9-]*)', comp):
                print(f'{os.path.relpath(f, root)}:{i}\t{hdr}\t{comp}')
```

`run_presence.sh`:

## H. ATDF pin signals and pinctrl header macros

`atdf_pins.py`:

```python
# P8.3: pin signals per peripheral from ATDF, and whether the pinctrl header of each port has a macro for each signal.
# usage: py atdf_pins.py <label> <atdf> <header>...
import sys, re, collections
import xml.etree.ElementTree as ET
label, atdf, *headers = sys.argv[1:]
mods = collections.OrderedDict()
tree = ET.parse(atdf)
for m in tree.getroot().iter('module'):
    for inst in m.iter('instance'):
        sig = inst.find('signals')
        if sig is None:
            continue
        for s in sig.iter('signal'):
            mods.setdefault((m.get('name'), inst.get('name')), []).append(
                (s.get('group'), s.get('index'), s.get('pad'), s.get('function')))
names = set()
for h in headers:
    for line in open(h, encoding='utf-8', errors='replace'):
        mm = re.match(r'#define\s+(P[A-D]\d+[A-Z]_\w+)', line)
        if mm:
            names.add(mm.group(1))
def macro(pad, fn, inst, grp, idx):
    p = re.match(r'P([A-D])(\d+)', pad)
    return f'P{p.group(1)}{int(p.group(2))}{fn}_{inst}_{grp}{idx if idx is not None else ""}'
print(f'### {label}: {atdf.split("/")[-1]}; headers: {[h.split("/")[-1] for h in headers]}; macros in headers: {len(names)}')
print('module\tinstance\tsignals\tgroups(count)\tmacro_found\tmacro_missing(first 3)')
WANT = {'TC','TCC','ADC','DAC','EIC','GMAC','RTC','WDT','TRNG','CAN','USB','DMAC','AC','SDHC','PTC','I2S','CCL','QSPI','PDEC'}
for (mod, inst), sigs in mods.items():
    if mod not in WANT:
        continue
    grp = collections.Counter(g for g, _, _, _ in sigs)
    found = miss = 0
    missing = []
    for g, i, pad, fn in sigs:
        mac = macro(pad, fn, inst, g, i)
        if mac in names:
            found += 1
        else:
            miss += 1
            missing.append(mac)
    print(f'{mod}\t{inst}\t{len(sigs)}\t{dict(grp)}\t{found}\t{miss}\t{missing[:3]}')
```

```python
# P8.3: dump the signals of one ATDF module instance, grouped by signal group. usage: py atdf_inst.py <atdf> <MODULE> <INSTANCE>...
import sys, collections
import xml.etree.ElementTree as ET
atdf, mod, *insts = sys.argv[1:]
root = ET.parse(atdf).getroot()
for m in root.iter('module'):
    if m.get('name') != mod:
        continue
    for inst in m.iter('instance'):
        if insts and inst.get('name') not in insts:
            continue
        g = collections.OrderedDict()
        for s in inst.iter('signal'):
            g.setdefault((s.get('group'), s.get('index')), []).append(f"{s.get('pad')}/{s.get('function')}" + (f"(io{s.get('ioset')})" if s.get('ioset') else ''))
        print(f'## {inst.get("name")} ({mod})')
        for (grp, idx), pads in g.items():
            print(f'  {grp}{idx if idx is not None else ""}: {", ".join(pads)}')
```

## J. Table generator and verifier

`gen.py` (reads `enum_src.txt`, `enum_tgt.txt`, `gen_data.py`; writes `tables.md`; asserts rows = enumerated total and referenced + target-only = target total):

```python
# P8.3 table generator: reads enum_src.txt / enum_tgt.txt (enumerator output) and gen_data.py, writes tables.md
import re, collections
from gen_data import PAIRS


def parse(path):
    d = collections.OrderedDict(); cur = None; lvl = None
    for line in open(path, encoding='utf-8'):
        line = line.rstrip('\n')
        m = re.match(r'## (\S+)\s+compatible=(\S+)', line)
        if m:
            cur = {'path': m.group(1), 'compat': m.group(2), 'levels': collections.OrderedDict(), 'total': 0}
            d[m.group(1).split('/')[-1]] = cur
            continue
        m = re.match(r'### level: (\S+)', line)
        if m:
            lvl = 'top' if m.group(1) == 'top' else 'child'
            cur['levels'][lvl] = []
            continue
        m = re.match(r'  (\S+)\ttype=(\S+)\trequired=(\S+)\t(.*)$', line)
        if m:
            cur['levels'][lvl].append(dict(name=m.group(1), type=m.group(2), req=m.group(3), origin=m.group(4)))
            continue
        m = re.match(r'TOTAL-DECLARED-PROPERTIES (\d+)', line)
        if m:
            cur['total'] = int(m.group(1))
    return d


SRC = parse('enum_src.txt')
TGT = parse('enum_tgt.txt')
# the 6.4 drift pair reads its source from the RTOS4 file and its target from main (same basename): keep both
SRC_DRIFT = parse('enum_src.txt')


def lvlname(l): return '' if l == 'top' else 'child: '


def origtag(o): return '' if o == 'own' else f' (inh: {o.replace("inh:", "")})'


def bt(s): return '`' + s + '`'


def prow(cells): return '| ' + ' | '.join(cells) + ' |'


out = []
report = []
for p in PAIRS:
    s = SRC[p['src']]
    t = TGT[p['tgt']] if p['tgt'] else None
    nrows = sum(len(v) for v in s['levels'].values())
    assert nrows == s['total'], (p['id'], nrows, s['total'])
    tmap = {}
    if t:
        for lv, props in t['levels'].items():
            for pr in props:
                tmap[(lv, pr['name'])] = pr
    referenced = set()
    rows = []
    for lv, props in s['levels'].items():
        for pr in props:
            key = (lv, pr['name'])
            ov = p['ov'].get(key)
            tname = None; mapping = None; note = None
            if ov is not None:
                tname, mapping, note = ov
            elif t is None:
                mapping = 'open'
                note = 'No target binding on main.'
            else:
                if key not in tmap:
                    raise SystemExit(f'MISSING OVERRIDE {p["id"]} {key} origin={pr["origin"]}')
                tname = pr['name']; mapping = 'keep'
                tp = tmap[key]
                inh = pr['origin'] != 'own'
                note = (f'Inherited from {pr["origin"].replace("inh:", "")}; same name and type exist on the target side.' if inh else 'Same name on the target.')
                if pr['type'] != '-' and tp['type'] != '-' and pr['type'] != tp['type']:
                    note += f' Type differs: {pr["type"]} vs {tp["type"]}.'
                if pr['req'] != tp['req']:
                    note += f' Required: {pr["req"]} on the source, {tp["req"]} on the target.'
            if tname is not None:
                if (lv, tname) not in tmap:
                    raise SystemExit(f'target prop not found {p["id"]} {key} -> {tname}')
                tp = tmap[(lv, tname)]
                referenced.add((lv, tname))
                tcell = bt(f'{p["tshort"]}: {lvlname(lv)}{tname}'); tt, tr = tp['type'], tp['req']
            else:
                tcell = '—'; tt = tr = '—'
            label = f'{lvlname(lv)}{pr["name"]}{origtag(pr["origin"])}'
            rows.append(prow([bt(label), pr['type'], pr['req'], tcell, tt, tr, mapping, note]))
    out += [f'### {p["id"]} {p["title"]}', '']
    out.append(f'Source: `{s["path"]}` (compatible `{s["compat"]}`).' + (f' Target: `{t["path"]}` (compatible `{t["compat"]}`).' if t else ' Target: none.'))
    out += ['', p['intro'], '']
    out.append(f'Source properties: **{s["total"]}** (own + inherited, incl. child-binding levels). Table rows: **{len(rows)}**.')
    out += ['', '| source property | type | required | target property | type | required | mapping | note |', '|---|---|---|---|---|---|---|---|']
    out += rows + ['']
    if t:
        tonly_rows = []
        for lv, props in t['levels'].items():
            for pr in props:
                if (lv, pr['name']) not in referenced:
                    note = p['tonly'].get(pr['name'] if lv == 'top' else 'child: ' + pr['name'])
                    if note is None:
                        note = 'Target-only (no source counterpart).'
                        if pr['origin'] != 'own':
                            note += f' Inherited from {pr["origin"].replace("inh:", "")}.'
                    tonly_rows.append(prow([bt(f'{p["tshort"]}: {lvlname(lv)}{pr["name"]}{origtag(pr["origin"])}'), pr['type'], pr['req'], note]))
        out.append(f'Target `{t["path"].split("/")[-1]}`: **{t["total"]}** properties; referenced by source rows: **{len(referenced)}**; target-only rows below: **{len(tonly_rows)}**.')
        out.append('')
        if tonly_rows:
            out += ['| target property | type | required | note |', '|---|---|---|---|'] + tonly_rows + ['']
        assert len(referenced) + len(tonly_rows) == t['total'], (p['id'], len(referenced), len(tonly_rows), t['total'])
    for v in p.get('variants', []):
        vt = TGT[v]
        a = {(lv, x['name']): (x['type'], x['req']) for lv, ps in t['levels'].items() for x in ps}
        b = {(lv, x['name']): (x['type'], x['req']) for lv, ps in vt['levels'].items() for x in ps}
        only_v = sorted(k[1] for k in set(b) - set(a)); only_p = sorted(k[1] for k in set(a) - set(b))
        diff = sorted(k[1] for k in set(a) & set(b) if a[k] != b[k])
        same = len(set(a) & set(b)) - len(diff)
        out.append(f'Variant `{vt["path"].split("/")[-1]}` (compatible `{vt["compat"]}`, **{vt["total"]}** properties) against `{t["path"].split("/")[-1]}` (**{t["total"]}**): equal in name, type and required flag: **{same}**; only in the variant: {only_v or "none"}; only in the primary target: {only_p or "none"}; same name, different type or required: {diff or "none"}.')
        out.append('')
        report.append((p['id'], v, vt['total'], same, only_v, only_p, diff))
open('tables.md', 'w', encoding='utf-8').write('\n'.join(out) + '\n')
print('tables.md written; pairs:', len(PAIRS))
for r in report:
    print(r)
```

## L. Recommendation table (copy of section 0 of the document)

"Recipe" and "generic form" are the two ways the configurator can offer a peripheral (CONFIGURATOR.md, "Three independent plugin kinds" and "Without a recipe ..."). A recipe is justified where the peripheral has pins the user must choose or where properties are derived from them. Everything else is served by the generic form, which needs only the binding.

|---|---|---|---|---|---|
| **TC, PWM** | `atmel,sam0-tc-pwm` | `microchip,tc-g1-pwm` | yes / yes (TC0-TC7 on E5x `_p`, only TC0-TC5 in `pic32cx_sg.dtsi`) | **Own recipe**, one module `timer_pwm` shared with TCC PWM | WO pads from the ATDF (function letter E/F/G depends on the pad); `#pwm-cells` 2 to 3 (consumer `pwms` need a third cell); `max-bit-width` 32 means pairing TCn+TCn+1 and a `client_mclk` clock; prescaler and period limits; GCLK source. |
| **TC, counter** | `atmel,sam0-tc32` | `microchip,tc-g1-counter` | yes / yes | **Generic form** | No pins. Only `prescaler` and, for 32 bit, `max-bit-width`. Translation needs a hook (32-bit pairing). |
