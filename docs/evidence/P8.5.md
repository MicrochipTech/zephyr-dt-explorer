`DTX_DESIGN_FILES=C:\Masters\board_design_files`. All work on a COPY of `_soltest\apps\lab1_leer` under
`%TEMP%\dtx_p85` (app and build folder). Drafts stay in memory (`ConfigSession.save()` is never called), nothing is
written into `apps\` or `_soltest\`. Verify with west runs in a scratch folder (`core/verify.py`).

## What was built

* `dt_explorer/core/generic.py`: form spec from the binding chain (`form_spec`: type, enum, default, required,
  description, current value), `check_form` (required / type / enum / boolean the board sets), `form_props` (only values
  that differ from the node), pin roles from the provider's signals (`pin_roles`, `pad_candidates`, `pin_problems`),
  `build_generic_node` (recipe-less `DraftNode`, class `generic`). The overlay is rendered by the unchanged
  `render_overlay`; `core/render.py` and `core/draft.py` needed nothing for recipe-less nodes.
* `dt_explorer/ui/generic_form.py`: the dialog (ids `gen-prop-<name>`, `gen-role-<role>`, `gen-problems`,
  `gen-save`, `gen-notes`, `gen-empty`); `ui/configure.py`: `can_configure` accepts a disabled node with a binding but no
  recipe, `open_configure` hands such a node to the generic dialog.
* Scope notes: the Configure... button condition in `ui/page.py` (`n.status == "disabled"`) is not in this card's
  write scope, so an enabled node without a recipe (`wdog`) gets no button; its generic draft is built through the core
  functions (below). A pin list is offered only if the binding declares `pinctrl-0` and the device model knows the instance.

```python
"""P8.5 acceptance: wdog and tc0 on a COPY of _soltest/apps/lab1_leer without a recipe."""
import shutil
import sys
import tempfile
from pathlib import Path

sys.path.insert(0, r"C:\Masters\dt_explorer\.claude\worktrees\agent-a04c555151faedd07")
from dt_explorer.app import load_plugins
from dt_explorer.core import generic as G
from dt_explorer.core import verify
from dt_explorer.core.build import BuildModel
from dt_explorer.ui import configure as C
from dt_explorer.ui.generic_form import current_values, form_state, generic_available

load_plugins()
src = Path(r"C:\Masters\26020_RTOS4\zephyrproject\_soltest\apps\lab1_leer")
work = Path(tempfile.gettempdir()) / "dtx_p85"
shutil.rmtree(work, ignore_errors=True)
app = work / "lab1_leer"
shutil.copytree(src, app)
build = app / "build_same54_xpro"
print("copy:", build)

m = BuildModel(build)
s = C.ConfigSession(m)           # drafts are kept in memory only: session.save() is never called
for label in ("wdog", "tc0"):
    n = s.node_for_label(label)
    compat, specs, roles, notes = form_state(s, n)
    print(f"\n== {label}: status={n.status} compatible={compat} recipe options={len(s.options_for(n))} "
          f"generic_available={generic_available(s, n)}")
    for f in specs:
        print(f"   field {f.name}: type={f.type} required={f.required} enum={f.enum} default={f.default} "
              f"current={f.current}  # {f.description}")
    print("   pin roles:", roles, "| not editable:", notes)

wdog = s.node_for_label("wdog")
tc0 = s.node_for_label("tc0")
s.upsert(G.build_generic_node(label="wdog", compatible=wdog.compats[0],
                              instance=s.instance_of(wdog).name if s.instance_of(wdog) else None))
_c, specs, _r, _n = form_state(s, tc0)
texts = {"prescaler": "64"}
assert not G.check_form(specs, texts)
s.upsert(G.build_generic_node(label="tc0", compatible=tc0.compats[0], instance=s.instance_of(tc0).name,
                              props=G.form_props(specs, texts)))
# a third node with a pad: tc2, first free candidate for WO0 (pins are optional for a counter; this proves the path)
tc2 = s.node_for_label("tc2")
_c, specs2, roles2, _n = form_state(s, tc2)
inst2 = s.instance_of(tc2)
used = s.used_by()
cands = G.pad_candidates(inst2, "WO0", {})
pad = next(p for p, mux in cands if not used.get(p))
print("tc2 WO0 candidates:", cands, "-> chosen", pad, "problems:", G.pin_problems(inst2, {"WO0": pad}))
s.upsert(G.build_generic_node(label="tc2", compatible=tc2.compats[0], instance=inst2.name, pins={"WO0": pad},
                              roles_order=roles2))
print("\ndraft nodes:", [(n.label, n.cls, n.compatible, n.props, n.pins) for n in s.draft.nodes])
overlay = s.overlay()
print("\n--- overlay ---\n" + overlay)
findings = s.check()
print("--- validate findings (%d) ---" % len(findings))
for f in findings:
    print(f.severity, f.kind, f.node, f.message)
print("errors:", [f for f in findings if f.severity == "error"])
from dt_explorer.ui.draft_tab import kconfig_lines
conf = "\n".join(kconfig_lines(s.kconfig())) + "\n"
print("--- kconfig suggestions (prj.conf lines) ---")
print(conf)
res = verify.verify(build, overlay, conf, on_line=None, scratch_root=work / "verify", timeout=600)
print("--- Verify with west ---")
print("rc:", res.returncode, "ok:", res.ok, "duration: %.1f s" % res.duration)
print("errors:", res.errors)
print("warnings:", res.warnings)
print("conf used:", repr(conf))
print("cmd:", " ".join(res.cmd))
print("log tail:\n" + "\n".join(res.log_tail[-12:]))
```
